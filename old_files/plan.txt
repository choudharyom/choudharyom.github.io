[X] Step 8.1: Create Article Template
    - Create folder: `src/templates`
    - Create file: `src/templates/post-template.js`
    - Add basic React component structure with GraphQL query:
      ```javascript
      import React from "react"
      import { graphql } from "gatsby"

      // Import Prism CSS (choose a theme)
      // You might need to install it if not included with prismjs, or find the path
      // e.g., require("prismjs/themes/prism-okaidia.css")
      // Alternatively, include CSS via gatsby-browser.js or layout component later.
      // For now, let's rely on gatsby-remark-prismjs default behavior or add CSS manually later.

      export default function PostTemplate({ data }) {
        const post = data.markdownRemark
        return (
          <div> {/* Basic wrapper, style later */}
            <h1>{post.frontmatter.title}</h1>
            <div dangerouslySetInnerHTML={{ __html: post.html }} />
          </div>
        )
      }

      export const query = graphql`
        query($slug: String!) {
          markdownRemark(frontmatter: { slug: { eq: $slug } }) {
            html
            frontmatter {
              title
              date(formatString: "MMMM DD, YYYY")
            }
          }
        }
      `
      ```
      *Note: KaTeX rendering requires its CSS to be included. `gatsby-remark-katex` handles this integration during build time.*
      *Note: PrismJS CSS needs to be included for highlighting to appear. You can add `require("prismjs/themes/prism-tomorrow.css");` (or another theme) at the top of `gatsby-browser.js`.*
